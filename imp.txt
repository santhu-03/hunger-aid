## Location-Based Volunteer Assignment Algorithm

```js
export function calculateHaversineDistance(lat1, lon1, lat2, lon2) {
  const R = 6371;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return Math.round(R * c * 100) / 100;
}
```

## Real-Time Route Polyline Rendering
This route fetcher decodes the Google polyline so the map can render the full route path.
```js
export async function fetchRoute(origin, destination, opts = {}) {
  const params = new URLSearchParams({
    origin: `${origin.latitude},${origin.longitude}`,
    destination: `${destination.latitude},${destination.longitude}`,
    mode: 'driving',
    key: GOOGLE_MAPS_KEY,
  });
  const json = await (await fetch(`https://maps.googleapis.com/maps/api/directions/json?${params}`)).json();
  const leg = json.routes[0].legs[0];
  const poly = json.routes[0].overview_polyline.points;
  const coords = decodePolyline(poly);

  return {
    polylineCoords: coords,
    encodedPolyline: poly,
    distanceMeters: leg.distance.value,
    durationSeconds: leg.duration.value,
    steps: leg.steps.map(s => ({
      instruction: s.html_instructions.replace(/<[^>]+>/g, ''),
      distanceMeters: s.distance.value,
      durationSeconds: s.duration.value,
    })),
  };
}
```

## ETA Calculation Function
This ETA helper prefers traffic-aware Google data and falls back to Haversine distance plus a default speed.
```js
export async function calculateETA(currentPos, destination, knownSpeedMs = null) {
  const distKm = haversineKm(
    currentPos.latitude, currentPos.longitude,
    destination.latitude, destination.longitude
  );

  if (GOOGLE_MAPS_KEY) {
    const matrix = await fetchDistanceMatrix([currentPos], [destination]);
    if (matrix?.[0]?.[0]?.durationSeconds != null) {
      const el = matrix[0][0];
      const secs = el.durationInTrafficSeconds ?? el.durationSeconds;
      return { etaMinutes: Math.max(1, Math.ceil(secs / 60)), distanceMeters: el.distanceMeters, source: 'google_matrix' };
    }
  }

  const speedKmh = knownSpeedMs != null && knownSpeedMs > 0.5 ? (knownSpeedMs * 3.6) : 20;
  return {
    etaMinutes: Math.max(1, Math.ceil((distKm / speedKmh) * 60)),
    distanceMeters: Math.round(distKm * 1000),
    source: 'haversine',
  };
}
```

## OTP Validation Logic
This validation flow prevents reuse, enforces expiry, and limits incorrect attempts before marking the delivery complete.
```js
export async function validateOTP(requestId, inputOtp) {
  const ref = doc(db, 'foodRequests', requestId);
  const snap = await getDoc(ref);
  const { otp, otpVerified, otpCreatedAt, otpAttempts = 0, otpExpired = false, otpCleared = false } = snap.data();

  if (otpVerified || otpCleared || !otp || otpExpired) return { valid: false, error: 'OTP is no longer valid for this request.' };
  if (otpCreatedAt && Date.now() - otpCreatedAt.toMillis() > OTP_EXPIRY_MS) {
    await updateDoc(ref, { otpExpired: true });
    return { valid: false, error: 'OTP expired. Ask the beneficiary for a new code.' };
  }
  if (otpAttempts >= MAX_OTP_ATTEMPTS) return { valid: false, error: 'Maximum attempts reached. Ask for a new OTP.', attemptsLeft: 0 };

  await updateDoc(ref, { otpAttempts: otpAttempts + 1 });
  if (String(otp).trim() !== String(inputOtp).trim()) return { valid: false, error: 'Incorrect OTP.' };

  await updateDoc(ref, { otpVerified: true, otpVerifiedAt: serverTimestamp(), status: 'Completed', otp: null, otpCleared: true });
  return { valid: true, attemptsLeft: MAX_OTP_ATTEMPTS - (otpAttempts + 1) };
}
```

## Secure Delivery Verification Workflow
This guard prevents unauthorized OTP resends and atomically replaces the code when a resend is allowed.
```js
export async function canResendOTP(requestId, volunteerId = null) {
  const snap = await getDoc(doc(db, 'foodRequests', requestId));
  const data = snap.data();
  if (volunteerId && data.volunteerId && data.volunteerId !== volunteerId) {
    return { allowed: false, reason: 'You are not assigned to this delivery.' };
  }
  if (data.otpVerified || data.status === 'Completed' || data.status === 'Completed Verified') {
    return { allowed: false, reason: 'Delivery is already completed.' };
  }
  if ((data.otpResendCount || 0) >= MAX_OTP_RESENDS) {
    return { allowed: false, reason: `Maximum ${MAX_OTP_RESENDS} resends reached.` };
  }
  return { allowed: true };
}
```

```js
export async function awardDeliveryPoints(volunteerId, volunteerName, { verified = false, onTime = false } = {}) {
  await initRewards(volunteerId, volunteerName);
  const ref = doc(db, 'rewards', volunteerId);
  const data = (await getDoc(ref)).data();

  const today = todayStr();
  const yesterday = yesterdayStr();
  const lastDate = data.lastDeliveryDate;
  let newStreak = 1;
  if (lastDate === yesterday) newStreak = (data.currentStreak || 0) + 1;
  else if (lastDate === today) newStreak = data.currentStreak || 1;

  let pts = verified ? POINTS.DELIVERY_VERIFIED : POINTS.DELIVERY_COMPLETED;
  if (onTime) pts += POINTS.ON_TIME_PICKUP;
  pts += Math.min(newStreak, 10) * POINTS.STREAK_BONUS;

  await updateDoc(ref, { totalPoints: increment(pts), currentStreak: newStreak, updatedAt: serverTimestamp() });
  return { pts, newStreak };
```

## Navigation Support Functions
These helper functions compute route deviation and provide a traffic-aware ETA for navigation screens.
```js
export function getRouteDeviation(currentPos, polylineCoords) {
  if (!polylineCoords?.length || !currentPos?.latitude) return Infinity;
  let minDist = Infinity;
  for (let i = 0; i < polylineCoords.length - 1; i++) {
    const d = _pointToSegmentDistance(currentPos, polylineCoords[i], polylineCoords[i + 1], 6_371_000);
    if (d < minDist) minDist = d;
  }
  return minDist;
}
```
